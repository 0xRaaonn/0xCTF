configure_prompt() {
    prompt_symbol=㉿
    # Skull emoji for root terminal
    #[ "$EUID" -eq 0 ] && prompt_symbol=💀
    case "$PROMPT_ALTERNATIVE" in
        twoline)
            PROMPT=$'%F{%(#.blue.green)}┌──${debian_chroot:+($debian_chroot)─}${VIRTUAL_ENV:+($(basename $VIRTUAL_ENV))─}(%B%F{%(#.red.blue)}%n'$prompt_symbol$'%m%b%F{%(#.blue.green)})-[%B%F{reset}%(6~.%-1~/…/%4~.%5~)%b%F{%(#.blue.green)}]\n└─%B%(#.%F{red}#.%F{blue}$)%b%F{reset} '
            # Right-side prompt with exit codes and background processes
            #RPROMPT=$'%(?.. %? %F{red}%B⨯%b%F{reset})%(1j. %j %F{yellow}%B⚙%b%F{reset}.)'
            ;;
        oneline)
            PROMPT=$'${debian_chroot:+($debian_chroot)}${VIRTUAL_ENV:+($(basename $VIRTUAL_ENV))}%B%F{%(#.red.blue)}%n@%m%b%F{reset}:%B%F{%(#.blue.green)}%~%b%F{reset}%(#.#.$) '
            RPROMPT=
            ;;
        backtrack)
            PROMPT=$'${debian_chroot:+($debian_chroot)}${VIRTUAL_ENV:+($(basename $VIRTUAL_ENV))}%B%F{red}%n@%m%b%F{reset}:%B%F{blue}%~%b%F{reset}%(#.#.$) '
            RPROMPT=
            ;;
    esac
    unset prompt_symbol
}

deletegit() {
  if [ -z "$1" ]; then
    echo "Usage: deletegit <file_path> [commit_message]"
    return 1
  fi

  local file=$1
  shift

  if [ ! -f "$file" ]; then
    echo "File '$file' does not exist."
    return 1
  fi

  git rm "$file"
  
  local msg=${1:-"deleted $file"}

  git commit -m "$msg"
  git push
}
deletegitfolder() {
  if [ -z "$1" ]; then
    echo "Usage: deletegitfolder <folder_path> [commit_message]"
    return 1
  fi

  local folder=$1
  shift

  if [ ! -d "$folder" ]; then
    echo "Folder '$folder' does not exist."
    return 1
  fi

  git rm -r "$folder"

  local msg=${1:-"deleted folder $folder"}

  git commit -m "$msg"
  git push
}

readme() {
    if [ -z "$1" ]; then
        echo "Usage: readme {-n|-a|-o|-s} \"description\""
        return 1
    fi

    flag="$1"
    # Combine all arguments after the flag for multi-word text
    text="${@:2}"

    case "$flag" in
        -n)
            if [ ! -f README.md ]; then
                echo "$text" > README.md
                echo "Created new README.md with: $text"
            else
                echo "⚠️ README.md already exists! Use -a to append or -o to overwrite."
            fi
            ;;
        -a)
            # Add a newline before appending
            echo "" >> README.md
            echo "$text" >> README.md
            echo "Appended to README.md: $text"
            ;;
        -o)
            echo "$text" > README.md
            echo "Overwrote README.md with: $text"
            ;;
        -s)
            # Open README.md in Sublime Text
            subl README.md
            ;;
        *)
            echo "Usage: readme {-n|-a|-o|-s} \"description\""
            echo "   -n   create new (fails if file exists)"
            echo "   -a   append (starts on a new line)"
            echo "   -o   overwrite"
            echo "   -s   open README.md in Sublime Text"
            ;;
    esac
}
flagsolve() {
    if [ -z "$1" ]; then
        echo "Usage: flagsolve \"text\""
        return 1
    fi

    # Combine all arguments to support multi-word text
    text="${*}"

    # Ensure we start on a new line, then add the bash block
    echo "" >> README.md
    echo '```bash' >> README.md
    echo "$text" >> README.md
    echo '```' >> README.md

    echo "Added flag to README.md in a bash block."
}
flagshow() {
    # Check if README.md exists
    if [ ! -f README.md ]; then
        echo "README.md does not exist!"
        return 1
    fi

    # Extract the last bash block from README.md
    output=$(awk '/```bash/{flag=1; next} /```/{flag=0} flag{line[NR]=$0} END{for(i in line) print line[i]}' README.md)

    if [ -z "$output" ]; then
        echo "Not yet inputted"
    else
        echo "$output"
    fi
}

complete() {
    # If called with no args → rename the current directory
    if [ $# -eq 0 ]; then
        current_dir=$(basename "$PWD")
        parent_dir=$(dirname "$PWD")

        # Check if already completed
        if [[ "$current_dir" == *_COMPLETED ]]; then
            echo "Folder '$current_dir' is already marked as completed."
            return 1
        fi

        new_name="${current_dir}_COMPLETED"
        new_path="${parent_dir}/${new_name}"

        if [ -d "$new_path" ]; then
            echo "Folder '$new_name' already exists!"
            return 1
        fi

        # Move up one directory and rename
        cd "$parent_dir" || { echo "Failed to cd into parent dir"; return 1; }
        mv "$current_dir" "$new_name"
        echo "Renamed '$current_dir' → '$new_name'"

        # Stay in parent folder
        return 0
    fi

    # If args are given → rename folders by name (like before)
    for folder in "$@"; do
        if [ ! -d "$folder" ]; then
            echo "Folder '$folder' does not exist!"
            continue
        fi

        if [[ "$folder" == *_COMPLETED ]]; then
            echo "Folder '$folder' is already marked as completed."
            continue
        fi

        new_name="${folder}_COMPLETED"
        if [ -d "$new_name" ]; then
            echo "Folder '$new_name' already exists!"
            continue
        fi

        mv "$folder" "$new_name"
        echo "Renamed '$folder' → '$new_name'"
    done
}
folder() {
    if [ -z "$1" ]; then
        echo "Usage: mkcd foldername"
        return 1
    fi

    if [ -d "$1" ]; then
        echo "Warning: Folder '$1' already exists."
    else
        mkdir "$1"
        echo "Folder '$1' created."
    fi

    cd "$1" || echo "Failed to enter folder '$1'"
}
